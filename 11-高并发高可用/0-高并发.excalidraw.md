```mermaid
flowchart TD

    end

    %% 应用核心服务
    subgraph AppLayer ["4. 应用业务服务集群 (无状态微服务)"]

    end

    %% 缓存与中间件
    subgraph MiddleLayer ["5. 缓存与消息中间件层"]

        MQCluster[("RocketMQ / Kafka 集群<br/>· 事务消息 (解耦下游履约)<br/>· 支付状态变更事件广播<br/>· 延迟消息 (未支付超时关单)")]
    end

    %% 数据存储与报表检索
    subgraph DataLayer ["6. 数据持久化与检索层"]
        ShardingDB[("MySQL 分库分表集群 (ShardingSphere)<br/>· 32 库 256 表 (按 user_id 分片)<br/>· 联合唯一索引防重<br/>· 仅存 3 个月在线热数据")]
        Canal["Canal / Debezium (监听 Binlog 增量)"]
        ES_CH[("检索与分析集群<br/>· Elasticsearch (商户多维检索)<br/>· ClickHouse (财务报表与对账聚合)")]
        ColdDB[("冷数据归档存储 (TiDB / HBase / 对象存储)<br/>· 存放 3 个月以上历史流水")]
    end

    %% 外部三方渠道
    subgraph ExternalTier ["7. 外部支付渠道与银行"]

        Alipay["支付宝 (Alipay)"]
        UnionPay["银联 / 网联 / 合作商业银行"]
    end

    %% 核心同步链路
    Client --> WAF
    WAF --> SLB
    SLB --> Nginx1
    SLB --> Nginx2
    Nginx1 --> Gateway
    Nginx2 --> Gateway

    Gateway -->|1. 校验防重/鉴权| RedisCluster
    Gateway -->|2. 转发核心支付指令| PayCore
    PayCore -->|3. 核心事务写入 & 查询| ShardingDB
    PayCore -->|4. 调用渠道统一指令| ChannelGw

    %% 渠道交互
    ChannelGw -->|HTTPS + 国密SM2/RSA2| WeChat
    ChannelGw -->|HTTPS + 专线/公网| Alipay
    ChannelGw -->|专线 ISO8583 / HTTP| UnionPay
    WeChat -.->|被动异步 Webhook 通知| ChannelGw
    Alipay -.->|被动异步 Webhook 通知| ChannelGw

    %% 异步解耦链路
    PayCore -->|5. 支付成功事务消息| MQCluster
    MQCluster -->|驱动下游| Downstream["下游系统 (订单履约 / 积分 / 商户结算)"]
    MQCluster -->|消费流水流式比对| ReconService

    %% 数据同步与归档
    ShardingDB -.-> Canal
    Canal -.-> ES_CH
    ShardingDB -.->|定时数据生命周期归档| ColdDB
    ReconService -->|拉取日终账单平账| ExternalTier
    ReconService -->|读取全量流水对账| ES_CH
```
